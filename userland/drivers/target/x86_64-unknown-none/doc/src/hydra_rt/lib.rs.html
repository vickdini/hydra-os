1//! # hydra-rt: runtime for Hydra's Ring 3 driver and server programs
2//!
3//! Driver programs are static native ELFs (no libc) that the supervisor starts in an
4//! address space of their own. This crate gives them what a C runtime would:
5//!
6//! - the process entry point (`_start`), arguments, a heap and a panic handler;
7//! - `print!`/`println!` to the console;
8//! - the native system calls ([`sys`]), I/O port access ([`port`]);
9//! - the server side of a driver service ([`Server`]): register, receive, reply, and the
10//!   shared data window.
11//!
12//! A program defines `fn main(args: &[&str])` and invokes [`entry!`] with it.
13

35/// Virtual size of the heap. It is reserved, not committed: pages are allocated on
36/// first use.
37const HEAP_SIZE: usize = 1024 * 1024 * 1024;
38
39#[global_allocator]
40static ALLOCATOR: linked_list_allocator::LockedHeap = linked_list_allocator::LockedHeap::empty();
41
42// The kernel enters with the stack holding argc, argv[] and envp[] (Linux layout)
43core::arch::global_asm!(
44    ".globl _start",
45    "_start:",
46    "mov rdi, rsp",
47    "and rsp, -16",
48    "call {start}",
49    "ud2",
50    start = sym rt_start,
51);
52
53extern "Rust" {
54    /// The program's `main`, exported by [`entry!`].
55    fn hydra_main(args: &[&'static str]);
56}
57
58/// Sets up the heap and the arguments, runs `main`, and exits.
59extern "C" fn rt_start(stack: *const u64) -> ! {
60    let heap = sys::map_anon(HEAP_SIZE).unwrap_or_else(|_| sys::exit(127));
61    unsafe { ALLOCATOR.lock().init(heap as *mut u8, HEAP_SIZE) };
62    let argc = unsafe { *stack } as usize;
63    let mut args: Vec<&'static str> = Vec::with_capacity(argc);
64    for i in 0..argc {
65        let p = unsafe { *stack.add(1 + i) } as *const u8;
66        let mut len = 0;
67        while unsafe { *p.add(len) } != 0 {
68            len += 1;
69        }
70        let bytes = unsafe { core::slice::from_raw_parts(p, len) };
71        args.push(core::str::from_utf8(bytes).unwrap_or(""));
72    }
73    unsafe { hydra_main(&args) };
74    sys::exit(0)
75}

77/// Exports `main` (a `fn(&[&str])`) as the program's entry point.
78#[macro_export]
79macro_rules! entry {
80    ($main:path) => {
81        #[no_mangle]
82        fn hydra_main(args: &[&'static str]) {
83            $main(args)
84        }
85    };
86}

88/// Heap bytes `(used, free)`.
89pub fn heap_usage() -> (usize, usize) {
90    let heap = ALLOCATOR.lock();
91    (heap.used(), heap.free())
92}

94/// Console output (standard output of the process, which the kernel prints).
95pub struct Console;
96
97impl Write for Console {
98    fn write_str(&mut self, s: &str) -> core::fmt::Result {
99        sys::write(1, s.as_bytes());
100        Ok(())
101    }
102}
103
104/// Prints to the console.
105#[macro_export]
106macro_rules! print {
107    ($($arg:tt)*) => {{
108        let _ = core::fmt::Write::write_fmt(&mut $crate::Console, format_args!($($arg)*));
109    }};
110}
111
112/// Prints a line to the console.
113#[macro_export]
114macro_rules! println {
115    () => { $crate::print!("\n") };
116    ($($arg:tt)*) => {{
117        // One write per line, so lines of different programs do not interleave
118        let line = alloc::format!("{}\n", format_args!($($arg)*));
119        $crate::sys::write(1, line.as_bytes());
120    }};
121}
122
123#[panic_handler]
124fn panic(info: &core::panic::PanicInfo) -> ! {
125    let mut line = alloc::string::String::new();
126    let _ = write!(line, "[driver] panic: {}\n", info);
127    sys::write(2, line.as_bytes());
128    sys::exit(101)
129}

131/// The server of a driver service: receives requests and answers them; bulk data is in
132/// the service window shared with the kernel.
133pub struct Server {
134    /// Endpoint served.
135    pub endpoint: u32,
136    window: *mut u8,
137    window_len: usize,
138}
139
140/// Size of a service window (as the kernel's `ipc::service::WINDOW_SIZE`).
141pub const WINDOW_SIZE: usize = 64 * 4096;
142
143impl Server {
144    /// Registers the calling process as the server of `endpoint` (the supervisor started
145    /// it for that service) and maps the service window.
146    pub fn register(endpoint: u32) -> Result<Self, i64> {
147        let window = sys::svc_register(endpoint)?;
148        Ok(Self { endpoint, window: window as *mut u8, window_len: WINDOW_SIZE })
149    }
150
151    /// The data window shared with clients.
152    pub fn window(&mut self) -> &mut [u8] {
153        unsafe { core::slice::from_raw_parts_mut(self.window, self.window_len) }
154    }
155
156    /// Serves requests forever: each request goes to `handle`, whose result is the reply.
157    /// Supervisor heartbeats are answered here.
158    pub fn serve(&mut self, mut handle: impl FnMut(&mut Self, &Message) -> Message) -> ! {
159        loop {
160            let req = match sys::svc_recv(self.endpoint) {
161                Ok(m) => m,
162                Err(_) => continue,
163            };
164            let reply = if req.label == protocol::MSG_SUPERVISOR_PING as u64 {
165                Message::new(protocol::MSG_SUPERVISOR_PONG as u64, [req.args[0], 1, 0, 0])
166            } else {
167                handle(self, &req)
168            };
169            let _ = sys::svc_reply(self.endpoint, &reply);
170        }
171    }
172}
