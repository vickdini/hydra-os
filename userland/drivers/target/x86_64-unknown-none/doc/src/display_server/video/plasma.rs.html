1//! # Video Subsystem: KDE Plasma 6.8 "Breeze" Desktop Shell
2//!
3//! Renders an authentic KDE Plasma 6.8 Wayland desktop environment complete with dark Breeze theme,
4//! floating translucent panel, Kickoff application menu, official Breeze SVG icons, and interactive window manager.
5//!
6//! ## Architectural Role & Design Rationale
7//!
8//! Hydra OS implements a desktop user interface modeling the KDE Plasma 6.8 workspace:
9//! - **Genuine KDE Plasma 6.8 Assets**: Downscaled upstream 5120x2880 dark ScarletTree wallpaper,
10//!   official Breeze icons (Konsole, KSysGuard, Dolphin, System Settings, KDE Plasma Kickoff).
11//! - **Breeze 6 Theme Engine**: Cohesive color palette utilizing Plasma blues (`#3daee9`, `#1d99f3`),
12//!   slate darks (`#1b1e20`, `#232629`, `#31363b`), and danger accents (`#da4453`).
13//! - **Floating Bottom Panel**: Plasma 6 signature floating dock detached 8px from bottom with
14//!   rounded corners, Kickoff launcher, active task tabs with focus indicators, and live system tray.
15//! - **Window Management**: Breeze window decorations, circular control buttons, titlebars,
16//!   and tear-free Wayland surface blitting.
17

54/// Draws a 32x32 RGBA/BGRA sprite with smooth alpha blending.
55pub fn draw_icon_32(canvas: &mut Compositor, x: usize, y: usize, icon: &[u8]) {
56    for iy in 0..32 {
57        for ix in 0..32 {
58            let idx = (iy * 32 + ix) * 4;
59            if idx + 3 < icon.len() {
60                let b = icon[idx];
61                let g = icon[idx + 1];
62                let r = icon[idx + 2];
63                let a = icon[idx + 3];
64                if a == 255 {
65                    canvas.put_pixel(x + ix, y + iy, Color::rgb(r, g, b));
66                } else if a > 0 {
67                    canvas.blend_pixel(x + ix, y + iy, Color::argb(a, r, g, b));
68                }
69            }
70        }
71    }
72}

74/// Draws a downsampled 16x16 icon for titlebars and task tabs.
75pub fn draw_icon_16(canvas: &mut Compositor, x: usize, y: usize, icon: &[u8]) {
76    for iy in 0..16 {
77        for ix in 0..16 {
78            let sx = ix * 2;
79            let sy = iy * 2;
80            let idx = (sy * 32 + sx) * 4;
81            if idx + 3 < icon.len() {
82                let b = icon[idx];
83                let g = icon[idx + 1];
84                let r = icon[idx + 2];
85                let a = icon[idx + 3];
86                if a > 0 {
87                    canvas.blend_pixel(x + ix, y + iy, Color::argb(a, r, g, b));
88                }
89            }
90        }
91    }
92}

94/// Renders the complete KDE Plasma desktop shell to the off-screen compositor canvas.
95///
96/// # Arguments
97/// - `canvas`: Double-buffered off-screen compositor canvas.
98/// - `compositor`: Active Wayland surface and window hierarchy.
99/// - `cursor_x`: Horizontal mouse pointer pixel coordinate.
100/// - `cursor_y`: Vertical mouse pointer pixel coordinate.
101/// - `frame_counter`: Monotonically increasing tick counter for subtle animations.
102pub fn render_plasma_desktop(
103    canvas: &mut Compositor,
104    compositor: &WaylandCompositor,
105    cursor_x: usize,
106    cursor_y: usize,
107    frame_counter: u64,
108) {
109    let width = canvas.width();
110    let height = canvas.height();
111
112    // 1. Render Official KDE Plasma 6.8 Wallpaper (ScarletTree)
113    render_wallpaper(canvas, width, height);
114
115    // 2. Render Managed Wayland Windows (in z-order from back to front)
116    for surface in &compositor.surfaces {
117        if surface.minimized {
118            continue;
119        }
120
121        render_window(canvas, surface, frame_counter);
122    }
123
124    // 3. Render Plasma 6 Floating Taskbar / Panel
125    render_panel(canvas, compositor, width, height, frame_counter);
126
127    // 4. Render Kickoff Application Menu (if toggled open)
128    if compositor.kickoff_open {
129        render_kickoff_menu(canvas, height);
130    }
131
132    // 5. Render Mouse Cursor Sprite
133    canvas.draw_cursor(cursor_x, cursor_y);
134}

136/// Renders the authentic KDE Plasma 6.8 ScarletTree wallpaper.
137fn render_wallpaper(canvas: &mut Compositor, width: usize, height: usize) {
138    if width == 1280 && height == 800 && SCARLETTREE_WALLPAPER.len() >= 1280 * 800 * 4 {
139        // Direct blit of authentic KDE Plasma 6.8 wallpaper
140        let src_u32: &[u32] = unsafe {
141            core::slice::from_raw_parts(
142                SCARLETTREE_WALLPAPER.as_ptr() as *const u32,
143                1280 * 800,
144            )
145        };
146        let target_len = src_u32.len().min(canvas.pixels_mut().len());
147        canvas.pixels_mut()[..target_len].copy_from_slice(&src_u32[..target_len]);
148    } else {
149        // Gradient fallback
150        canvas.draw_gradient_v(0, 0, width, height / 2, COLOR_WALLPAPER_TOP, COLOR_WALLPAPER_MID);
151        canvas.draw_gradient_v(0, height / 2, width, height - (height / 2), COLOR_WALLPAPER_MID, COLOR_WALLPAPER_BOT);
152    }
153
154    // Subtle watermark in bottom-right corner
155    canvas.draw_string(
156        width.saturating_sub(290),
157        height.saturating_sub(20),
158        "Hydra OS | KDE Plasma 6.8 (Breeze 6)",
159        COLOR_TEXT_MUTED,
160        None,
161    );
162}

164/// Renders an individual Wayland application surface and its Breeze window frame.
165fn render_window(canvas: &mut Compositor, surface: &crate::video::wayland::WlSurface, frame_counter: u64) {
166    let x = surface.x;
167    let y = surface.y;
168    let w = surface.width;
169    let h = surface.height;
170
171    // Window shadow (subtle dark translucent border)
172    canvas.draw_rect_outline(x.saturating_sub(2), y.saturating_sub(2), w + 4, h + 4, Color::rgb(10, 12, 14), 2);
173
174    // Window border
175    let border_color = if surface.focused { COLOR_PLASMA_BLUE } else { COLOR_WINDOW_BORDER };
176    canvas.draw_rect_outline(x, y, w, h, border_color, 1);
177
178    // Titlebar header (height 32px)
179    let titlebar_color = if surface.focused { COLOR_TITLEBAR_ACTIVE } else { COLOR_TITLEBAR_INACTIVE };
180    canvas.fill_rect(x, y, w, TITLEBAR_HEIGHT, titlebar_color);
181
182    // Titlebar top highlight line
183    canvas.fill_rect_alpha(x + 1, y + 1, w - 2, 1, Color::rgb(255, 255, 255), 30);
184
185    // Titlebar bottom separator line
186    canvas.fill_rect(x, y + TITLEBAR_HEIGHT - 1, w, 1, COLOR_PANEL_BORDER);
187
188    // Authentic Breeze Window Icon (16x16)
189    let win_icon = match surface.window_type {
190        WindowType::Konsole => ICON_TERMINAL,
191        WindowType::KSysGuard => ICON_SYS_MONITOR,
192        WindowType::WaylandClient => ICON_START_KDE,
193        WindowType::About => ICON_SETTINGS,
194    };
195    draw_icon_16(canvas, x + 8, y + 6, win_icon);
196
197    // Window title text
198    canvas.draw_string(x + 30, y + 6, &surface.title, COLOR_TEXT_LIGHT, None);
199
200    // Breeze 6 Window control buttons (Minimize, Maximize, Close) on top-right
201    let right_edge = x + w;
202
203    // Minimize button ('_')
204    let min_x = right_edge.saturating_sub(CONTROL_BTN_WIDTH * 3);
205    canvas.fill_rect(min_x, y, CONTROL_BTN_WIDTH, TITLEBAR_HEIGHT, titlebar_color);
206    canvas.draw_char(min_x + 10, y + 6, '-', COLOR_TEXT_MUTED, None);
207
208    // Maximize button ('#')
209    let max_x = right_edge.saturating_sub(CONTROL_BTN_WIDTH * 2);
210    canvas.fill_rect(max_x, y, CONTROL_BTN_WIDTH, TITLEBAR_HEIGHT, titlebar_color);
211    canvas.draw_char(max_x + 10, y + 6, if surface.maximized { '=' } else { '#' }, COLOR_TEXT_MUTED, None);
212
213    // Close button ('X') with red accent on hover/active
214    let close_x = right_edge.saturating_sub(CONTROL_BTN_WIDTH);
215    canvas.fill_rect(close_x, y, CONTROL_BTN_WIDTH, TITLEBAR_HEIGHT, COLOR_CLOSE_RED);
216    canvas.draw_char(close_x + 10, y + 6, 'X', COLOR_TEXT_LIGHT, None);
217
218    // Window client content area
219    let client_y = y + TITLEBAR_HEIGHT;
220    let client_h = h.saturating_sub(TITLEBAR_HEIGHT);
221
222    match surface.window_type {
223        WindowType::Konsole => {
224            render_konsole_content(canvas, surface, x, client_y, w, client_h, frame_counter);
225        }
226        WindowType::KSysGuard => {
227            render_ksysguard_content(canvas, x, client_y, w, client_h, frame_counter);
228        }
229        WindowType::About => {
230            canvas.fill_rect(x, client_y, w, client_h, COLOR_KSYSGUARD_BG);
231            canvas.draw_string(x + 20, client_y + 20, "Hydra OS Plasma Shell v0.1.0 (KDE 6.8)", COLOR_TEXT_LIGHT, None);
232        }
233        WindowType::WaylandClient => {
234            render_wayland_client_content(canvas, x, client_y, w, client_h, frame_counter);
235        }
236    }
237}

289/// Renders the Konsole terminal application content.
290fn render_konsole_content(
291    canvas: &mut Compositor,
292    surface: &crate::video::wayland::WlSurface,
293    x: usize,
294    y: usize,
295    w: usize,
296    h: usize,
297    frame_counter: u64,
298) {
299    canvas.fill_rect(x, y, w, h, COLOR_KONSOLE_BG);
300
301    let mut text_y = y + 8;
302    for line in &surface.terminal_lines {
303        if text_y + FONT_HEIGHT > y + h - 24 {
304            break;
305        }
306        canvas.draw_string(x + 12, text_y, line, COLOR_TEXT_LIGHT, None);
307        text_y += FONT_HEIGHT + 2;
308    }
309
310    // Active shell prompt and cursor
311    if text_y + FONT_HEIGHT <= y + h {
312        let prompt = "hydra:desktop> ";
313        canvas.draw_string(x + 12, text_y, prompt, COLOR_PLASMA_BLUE, None);
314        let prompt_width = prompt.len() * 8;
315
316        canvas.draw_string(x + 12 + prompt_width, text_y, &surface.terminal_input, COLOR_TEXT_LIGHT, None);
317
318        // Blinking cursor block
319        if (frame_counter / 15) % 2 == 0 {
320            let cursor_offset = (prompt.len() + surface.terminal_input.len()) * 8;
321            canvas.draw_char(x + 12 + cursor_offset, text_y, '#', COLOR_PLASMA_BLUE, None);
322        }
323    }
324}

382/// Renders the signature KDE Plasma 6.8 floating bottom taskbar panel.
383fn render_panel(
384    canvas: &mut Compositor,
385    compositor: &WaylandCompositor,
386    width: usize,
387    height: usize,
388    _frame_counter: u64,
389) {
390    // KDE Plasma 6 Floating Panel: 44px tall, detached from bottom by 8px, 16px margins
391    let panel_h = 44;
392    let panel_y = height.saturating_sub(panel_h + 8);
393    let panel_x = 16;
394    let panel_w = width.saturating_sub(32);
395
396    // Floating translucent glassmorphism background (#1b1e20ee)
397    canvas.fill_rect_alpha(panel_x, panel_y, panel_w, panel_h, COLOR_PANEL_BG, 235);
398    // 1px subtle border
399    canvas.draw_rect_outline(panel_x, panel_y, panel_w, panel_h, COLOR_PANEL_BORDER, 1);
400    // Subtle top highlight line
401    canvas.fill_rect_alpha(panel_x + 1, panel_y + 1, panel_w - 2, 1, Color::rgb(255, 255, 255), 45);
402
403    // Kickoff Application Launcher Button (far left of panel)
404    let kickoff_x = panel_x + 6;
405    let kickoff_y = panel_y + 4;
406    let kickoff_w = 42;
407    let kickoff_h = 36;
408    let kickoff_bg = if compositor.kickoff_open { COLOR_PLASMA_DEEP_BLUE } else { Color::rgb(38, 43, 48) };
409    canvas.fill_rect(kickoff_x, kickoff_y, kickoff_w, kickoff_h, kickoff_bg);
410    canvas.draw_rect_outline(
411        kickoff_x,
412        kickoff_y,
413        kickoff_w,
414        kickoff_h,
415        if compositor.kickoff_open { COLOR_PLASMA_BLUE } else { Color::rgb(55, 62, 69) },
416        1,
417    );
418
419    // Draw genuine 32x32 KDE Plasma Kickoff Icon
420    draw_icon_32(canvas, kickoff_x + 5, kickoff_y + 2, ICON_START_KDE);
421
422    // Taskbar entries for active open windows
423    let mut task_x = kickoff_x + kickoff_w + 12;
424    for surface in &compositor.surfaces {
425        let task_w = 170;
426        if task_x + task_w > width.saturating_sub(280) {
427            break;
428        }
429
430        let is_active = surface.focused && !surface.minimized;
431        let task_bg = if is_active {
432            Color::rgb(44, 50, 56)
433        } else {
434            Color::rgb(30, 34, 38)
435        };
436        canvas.fill_rect(task_x, panel_y + 5, task_w, 34, task_bg);
437        canvas.draw_rect_outline(task_x, panel_y + 5, task_w, 34, Color::rgb(52, 59, 66), 1);
438
439        // Highlight line on bottom of focused task (Breeze Blue)
440        if is_active {
441            canvas.fill_rect(task_x + 4, panel_y + panel_h - 7, task_w - 8, 3, COLOR_PLASMA_BLUE);
442        }
443
444        // Draw application icon
445        let icon = match surface.window_type {
446            WindowType::Konsole => ICON_TERMINAL,
447            WindowType::KSysGuard => ICON_SYS_MONITOR,
448            WindowType::WaylandClient => ICON_START_KDE,
449            WindowType::About => ICON_SETTINGS,
450        };
451        draw_icon_16(canvas, task_x + 8, panel_y + 14, icon);
452
453        let title_slice = if surface.title.len() > 15 { &surface.title[..15] } else { &surface.title };
454        canvas.draw_string(task_x + 30, panel_y + 14, title_slice, COLOR_TEXT_LIGHT, None);
455
456        task_x += task_w + 6;
457    }
458
459    // System Tray (right side of panel)
460    let tray_x = panel_x + panel_w - 260;
461
462    // ZFS indicator badge
463    canvas.fill_rect(tray_x, panel_y + 8, 100, 28, Color::rgb(32, 37, 41));
464    canvas.draw_rect_outline(tray_x, panel_y + 8, 100, 28, Color::rgb(48, 55, 62), 1);
465    canvas.fill_rect(tray_x + 8, panel_y + 19, 6, 6, COLOR_GREEN_ACCENT);

475/// Renders the Kickoff Application Menu popup above the launcher button.
476fn render_kickoff_menu(canvas: &mut Compositor, screen_height: usize) {
477    let menu_w = 280;
478    let menu_h = 260;
479    let menu_x = 16;
480    let menu_y = (screen_height.saturating_sub(52)).saturating_sub(menu_h + 8);
481
482    // Menu container & outline
483    canvas.fill_rect_alpha(menu_x, menu_y, menu_w, menu_h, COLOR_PANEL_BG, 240);
484    canvas.draw_rect_outline(menu_x, menu_y, menu_w, menu_h, COLOR_PLASMA_BLUE, 1);
485
486    // Top highlight line
487    canvas.fill_rect_alpha(menu_x + 1, menu_y + 1, menu_w - 2, 1, Color::rgb(255, 255, 255), 40);
488
489    // Search header
490    canvas.fill_rect(menu_x + 10, menu_y + 10, menu_w - 20, 28, COLOR_TITLEBAR_ACTIVE);
491    canvas.draw_rect_outline(menu_x + 10, menu_y + 10, menu_w - 20, 28, COLOR_PANEL_BORDER, 1);
492    canvas.draw_string(menu_x + 20, menu_y + 16, "Search applications...", COLOR_TEXT_MUTED, None);
493
494    // Menu Items with authentic Breeze icons
495    let items = [
496        ("Konsole (Terminal)", ICON_TERMINAL),
497        ("KSysGuard (Monitor)", ICON_SYS_MONITOR),
498        ("Dolphin (Files)", ICON_FILE_MANAGER),
499        ("System Settings", ICON_SETTINGS),
500        ("Leave / Exit (Esc)", ICON_START_KDE),
501    ];
502
503    let mut item_y = menu_y + 48;
504    for (i, (item, icon)) in items.iter().enumerate() {
505        let is_hover = i == 0;
506        if is_hover {
507            canvas.fill_rect(menu_x + 6, item_y, menu_w - 12, 34, COLOR_TITLEBAR_ACTIVE);
508            canvas.draw_rect_outline(menu_x + 6, item_y, menu_w - 12, 34, COLOR_PLASMA_BLUE, 1);
509        }
510
511        draw_icon_16(canvas, menu_x + 14, item_y + 8, icon);
512        canvas.draw_string(menu_x + 40, item_y + 10, item, COLOR_TEXT_LIGHT, None);
513        item_y += 38;
514    }
515}
