1//! # Video Subsystem: Microkernel Wayland Surface & Window Compositor
2//!
3//! Provides the core Wayland surface abstraction, window hierarchy, and input routing.
4//!
5//! ## Architectural Role & Design Rationale
6//!
7//! In standard Wayland architectures, clients allocate pixel buffers and attach them to
8//! `wl_surface` objects. The compositor handles window decorations (or accepts client-side decorations),
9//! manages the window stacking order (z-index), processes input focus (`wl_seat`), and composites
10//! surfaces into the display raster.
11//!
12//! This module implements the internal microkernel representation of Wayland surfaces,
13//! top-level application windows (`xdg_toplevel`), and desktop window interaction (movement,
14//! focusing, minimize, maximize, and close).
15//!
16//! ## Key Types & Interfaces
17//!
18//! - `WlSurface`: Represents an individual window surface with title, bounds, and content state.
19//! - `WindowType`: Supported desktop application types (Konsole Terminal, KSysGuard Monitor, About).
20//! - `WaylandCompositor`: Central registry managing surface lifecycle, z-ordering, and drag state.
21

27/// Standard height of the window title bar in pixels.
28pub const TITLEBAR_HEIGHT: usize = 28;

30/// Width of the Breeze window control buttons (Close, Maximize, Minimize).
31pub const CONTROL_BTN_WIDTH: usize = 28;

33/// Category of application surface rendered inside the Wayland window.
34#[derive(Debug, Clone, Copy, PartialEq, Eq)]
35pub enum WindowType {
36    /// Interactive Konsole terminal window executing Hydra microkernel shell commands.
37    Konsole,
38    /// KSysGuard system monitor displaying CPU, memory, and ZFS metrics.
39    KSysGuard,
40    /// Hydra OS & KDE Plasma desktop information dialog.
41    About,
42    /// Native Linux Wayland Client application.
43    WaylandClient,
44}

46/// A desktop application window backed by a Wayland surface.
47pub struct WlSurface {
48    /// Unique surface identifier.
49    pub id: u32,
50    /// Window title displayed in the title bar.
51    pub title: String,
52    /// Desktop window application type.
53    pub window_type: WindowType,
54    /// Current horizontal screen coordinate in pixels.
55    pub x: usize,
56    /// Current vertical screen coordinate in pixels.
57    pub y: usize,
58    /// Current window width in pixels.
59    pub width: usize,
60    /// Current window height in pixels.
61    pub height: usize,
62    /// True if the window holds input focus.
63    pub focused: bool,
64    /// True if the window is currently minimized to the task panel.
65    pub minimized: bool,
66    /// True if the window is maximized to fill the desktop canvas.
67    pub maximized: bool,
68    /// Restored coordinates `(x, y, width, height)` when un-maximizing.
69    pub saved_rect: (usize, usize, usize, usize),
70    /// True if the user is actively dragging the title bar.
71    pub dragging: bool,
72    /// Horizontal cursor offset within the title bar during drag.
73    pub drag_offset_x: usize,
74    /// Vertical cursor offset within the title bar during drag.
75    pub drag_offset_y: usize,
76    /// Active keyboard input line for embedded Konsole terminals.
77    pub terminal_input: String,
78    /// Scrollback text lines for embedded Konsole terminals.
79    pub terminal_lines: Vec<String>,
80}

82impl WlSurface {
83    /// Creates a new Wayland application surface.
84    pub fn new(
85        id: u32,
86        title: &str,
87        window_type: WindowType,
88        x: usize,
89        y: usize,
90        width: usize,
91        height: usize,
92    ) -> Self {
93        let mut terminal_lines = Vec::new();
94        if window_type == WindowType::Konsole {
95            terminal_lines.push(String::from("Hydra OS KDE Plasma 6.8 Desktop — Konsole v0.1.0"));
96            terminal_lines.push(String::from("Connected to Wayland Compositor (/run/user/1000/wayland-0)"));
97            terminal_lines.push(String::from("Linux ABI 6.8.0-hydra active (Dynamic Linker & wl_seat enabled)"));
98            terminal_lines.push(String::from("Type 'help', 'linux status', 'zpool', or 'mem'. Press 'Esc' to exit."));
99            terminal_lines.push(String::from(""));
100        }
101
102        Self {
103            id,
104            title: String::from(title),
105            window_type,
106            x,
107            y,
108            width,
109            height,
110            focused: false,
111            minimized: false,
112            maximized: false,
113            saved_rect: (x, y, width, height),
114            dragging: false,
115            drag_offset_x: 0,
116            drag_offset_y: 0,
117            terminal_input: String::new(),
118            terminal_lines,
119        }
120    }
121
122    /// Appends a line of text to the terminal output buffer.
123    pub fn append_terminal_line(&mut self, line: &str) {
124        if self.terminal_lines.len() >= 18 {
125            self.terminal_lines.remove(0);
126        }
127        self.terminal_lines.push(String::from(line));
128    }
129
130    /// Returns true if the coordinate `(px, py)` falls within the window boundaries.
131    pub fn contains_point(&self, px: usize, py: usize) -> bool {
132        if self.minimized {
133            return false;
134        }
135        px >= self.x && px < self.x + self.width && py >= self.y && py < self.y + self.height
136    }
137
138    /// Returns true if the coordinate `(px, py)` falls within the top title bar header.
139    pub fn is_in_titlebar(&self, px: usize, py: usize) -> bool {
140        if self.minimized {
141            return false;
142        }
143        px >= self.x
144            && px < self.x + self.width
145            && py >= self.y
146            && py < self.y + TITLEBAR_HEIGHT
147    }
148
149    /// Returns which title bar control button was clicked, if any.
150    ///
151    /// # Returns
152    /// - `Some("close")`: Close button clicked.
153    /// - `Some("maximize")`: Maximize/Restore button clicked.
154    /// - `Some("minimize")`: Minimize button clicked.
155    /// - `None`: Titlebar drag area or outside buttons.
156    pub fn check_control_buttons(&self, px: usize, py: usize) -> Option<&'static str> {
157        if !self.is_in_titlebar(px, py) {
158            return None;
159        }
160
161        let right_edge = self.x + self.width;
162        if px >= right_edge.saturating_sub(CONTROL_BTN_WIDTH) && px < right_edge {
163            Some("close")
164        } else if px >= right_edge.saturating_sub(CONTROL_BTN_WIDTH * 2)
165            && px < right_edge.saturating_sub(CONTROL_BTN_WIDTH)
166        {
167            Some("maximize")
168        } else if px >= right_edge.saturating_sub(CONTROL_BTN_WIDTH * 3)
169            && px < right_edge.saturating_sub(CONTROL_BTN_WIDTH * 2)
170        {
171            Some("minimize")
172        } else {
173            None
174        }
175    }
176}

178/// Central registry managing all active Wayland surfaces, z-stacking, and input focus.
179pub struct WaylandCompositor {
180    /// Active desktop surfaces ordered from bottom to top (z-order).
181    pub surfaces: Vec<WlSurface>,
182    /// Monotonically increasing surface identifier counter.
183    pub next_surface_id: u32,
184    /// Identifier of the surface currently being dragged by pointer, if any.
185    pub active_dragging_id: Option<u32>,
186    /// State of the KDE Plasma Kickoff start menu (open/closed).
187    pub kickoff_open: bool,
188}
189
190impl WaylandCompositor {
191    /// Creates a new compositor registry populated with default desktop applications.
192    pub fn new(screen_width: usize, screen_height: usize) -> Self {
193        let mut comp = Self {
194            surfaces: Vec::new(),
195            next_surface_id: 1,
196            active_dragging_id: None,
197            kickoff_open: false,
198        };
199
200        // Window 1: Konsole Terminal (Top-left)
201        let konsole_w = 540.min(screen_width.saturating_sub(40));
202        let konsole_h = 320.min(screen_height.saturating_sub(80));
203        let konsole_x = 36;
204        let konsole_y = 36;
205
206        let konsole = WlSurface::new(
207            1,
208            "Konsole — Hydra OS Terminal",
209            WindowType::Konsole,
210            konsole_x,
211            konsole_y,
212            konsole_w,
213            konsole_h,
214        );
215
216        // Window 2: KSysGuard System Monitor (Top-right)
217        let monitor_w = 440.min(screen_width.saturating_sub(40));
218        let monitor_h = 320.min(screen_height.saturating_sub(80));
219        let monitor_x = screen_width.saturating_sub(monitor_w + 36);
220        let monitor_y = 36;
221
222        let monitor = WlSurface::new(
223            2,
224            "KSysGuard — Plasma System Monitor",
225            WindowType::KSysGuard,
226            monitor_x,
227            monitor_y,
228            monitor_w,
229            monitor_h,
230        );
231
232        // Window 3: Plasma 6.8 Native Linux Wayland Client (Bottom-center)
233        let client_w = 520.min(screen_width.saturating_sub(40));
234        let client_h = 330.min(screen_height.saturating_sub(80));
235        let client_x = (screen_width.saturating_sub(client_w)) / 2;
236        let client_y = screen_height.saturating_sub(client_h + 52);
237
238        let wl_client = WlSurface::new(
239            3,
240            "Plasma 6.8 — Linux Wayland Client",
241            WindowType::WaylandClient,
242            client_x,
243            client_y,
244            client_w,
245            client_h,
246        );
247
248        comp.surfaces.push(wl_client);
249        comp.surfaces.push(monitor);
250        comp.surfaces.push(konsole); // Konsole on top initially
251        comp.set_focus(1);
252        comp.next_surface_id = 4;
253
254        comp
255    }

257    /// Sets keyboard and active focus to the surface with the specified ID.
258    pub fn set_focus(&mut self, id: u32) {
259        for s in &mut self.surfaces {
260            s.focused = s.id == id;
261        }
262    }
263
264    /// Brings the specified surface to the top of the z-order stack.
265    pub fn raise_to_top(&mut self, id: u32) {
266        if let Some(pos) = self.surfaces.iter().position(|s| s.id == id) {
267            let surface = self.surfaces.remove(pos);
268            self.surfaces.push(surface);
269            self.set_focus(id);
270        }
271    }
272
273    /// Returns the topmost surface under coordinate `(px, py)`, if any.
274    pub fn surface_at_point(&self, px: usize, py: usize) -> Option<u32> {
275        // Iterate backwards from topmost surface to bottom
276        for s in self.surfaces.iter().rev() {
277            if s.contains_point(px, py) {
278                return Some(s.id);
279            }
280        }
281        None
282    }

284    /// Dispatches a mouse press event to windows, title bars, and control buttons.
285    pub fn handle_mouse_down(
286        &mut self,
287        px: usize,
288        py: usize,
289        screen_width: usize,
290        screen_height: usize,
291    ) {
292        // Check if clicked in bottom floating panel kickoff launcher button
293        let panel_y = screen_height.saturating_sub(52);
294        if py >= panel_y {
295            if px >= 16 && px <= 76 {
296                self.kickoff_open = !self.kickoff_open;
297            } else {
298                self.kickoff_open = false;
299            }
300            return;
301        }
302
303        // If kickoff menu is open and click was outside, close it
304        if self.kickoff_open {
305            self.kickoff_open = false;
306        }
307
308        // Check window clicks in z-order
309        if let Some(target_id) = self.surface_at_point(px, py) {
310            self.raise_to_top(target_id);
311
312            // Re-find the raised surface (now at the back of `surfaces`)
313            if let Some(surface) = self.surfaces.iter_mut().find(|s| s.id == target_id) {
314                if let Some(btn) = surface.check_control_buttons(px, py) {
315                    match btn {
316                        "close" => {
317                            surface.minimized = true;
318                        }
319                        "maximize" => {
320                            if surface.maximized {
321                                // Restore
322                                let (ox, oy, ow, oh) = surface.saved_rect;
323                                surface.x = ox;
324                                surface.y = oy;
325                                surface.width = ow;
326                                surface.height = oh;
327                                surface.maximized = false;
328                            } else {
329                                // Maximize
330                                surface.saved_rect = (surface.x, surface.y, surface.width, surface.height);
331                                surface.x = 0;
332                                surface.y = 0;
333                                surface.width = screen_width;
334                                surface.height = screen_height.saturating_sub(40);
335                                surface.maximized = true;
336                            }
337                        }
338                        "minimize" => {
339                            surface.minimized = true;
340                        }
341                        _ => {}
342                    }
343                    return;
344                }
345
346                if surface.is_in_titlebar(px, py) && !surface.maximized {
347                    surface.dragging = true;
348                    surface.drag_offset_x = px.saturating_sub(surface.x);
349                    surface.drag_offset_y = py.saturating_sub(surface.y);
350                    self.active_dragging_id = Some(target_id);
351                }
352            }
353        }
354    }

356    /// Dispatches mouse movement updates for active window dragging.
357    pub fn handle_mouse_move(
358        &mut self,
359        px: usize,
360        py: usize,
361        screen_width: usize,
362        screen_height: usize,
363    ) {
364        if let Some(drag_id) = self.active_dragging_id {
365            if let Some(surface) = self.surfaces.iter_mut().find(|s| s.id == drag_id) {
366                if surface.dragging {
367                    let max_x = screen_width.saturating_sub(60);
368                    let max_y = (screen_height.saturating_sub(40)).saturating_sub(TITLEBAR_HEIGHT);
369
370                    let new_x = px.saturating_sub(surface.drag_offset_x).min(max_x);
371                    let new_y = py.saturating_sub(surface.drag_offset_y).min(max_y);
372
373                    surface.x = new_x;
374                    surface.y = new_y;
375                }
376            }
377        }
378    }
379
380    /// Dispatches mouse button release events and terminates active drags.
381    pub fn handle_mouse_up(&mut self) {
382        if let Some(drag_id) = self.active_dragging_id {
383            if let Some(surface) = self.surfaces.iter_mut().find(|s| s.id == drag_id) {
384                surface.dragging = false;
385            }
386        }
387        self.active_dragging_id = None;
388    }

390    /// Sends typed keyboard character input to the active focused window.
391    pub fn handle_key_char(&mut self, c: char) {
392        for surface in &mut self.surfaces {
393            if surface.focused && !surface.minimized && surface.window_type == WindowType::Konsole {
394                match c {
395                    '\r' | '\n' => {
396                        let cmd = surface.terminal_input.clone();
397                        surface.append_terminal_line(&alloc::format!("hydra:desktop> {}", cmd));
398                        surface.terminal_input.clear();
399                        execute_terminal_command(surface, &cmd);
400                    }
401                    '\x08' => {
402                        surface.terminal_input.pop();
403                    }
404                    printable => {
405                        if surface.terminal_input.len() < 64 && !printable.is_control() {
406                            surface.terminal_input.push(printable);
407                        }
408                    }
409                }
410                break;
411            }
412        }
413    }
414}

416/// Executes a shell command entered in the Konsole window and appends the result.
417fn execute_terminal_command(surface: &mut WlSurface, cmd: &str) {
418    let trimmed = cmd.trim();
419    if trimmed.is_empty() {
420        return;
421    }
422
423    match trimmed {
424        "help" => {
425            surface.append_terminal_line("Built-in GUI Commands: help, uname, zpool, mem, clear, exit");
426        }
427        "uname" => {
428            surface.append_terminal_line("Hydra OS 0.1.0 x86_64-unknown-uefi Microkernel (SMP/CoW)");
429        }
430        "zpool" => {
431            surface.append_terminal_line("pool: hydrapool | state: ONLINE | vdev-ata-0 (Persistent ATA)");
432            surface.append_terminal_line("capacity: 32 MiB | CoW Merkle Tree | Fletcher-4 verified");
433        }
434        "mem" => {
435            let heap_stats = crate::memory::heap::stats();
436            let free_mb = heap_stats.free_bytes / (1024 * 1024);
437            let total_mb = heap_stats.total_bytes / (1024 * 1024);
438            surface.append_terminal_line(&alloc::format!("Kernel Heap: {} MiB free / {} MiB total", free_mb, total_mb));
439        }
440
441        "clear" => {
442            surface.terminal_lines.clear();
443        }
444        "exit" => {
445            surface.append_terminal_line("Use 'Esc' key on keyboard to return to text console.");
446        }
447        cmd if cmd.starts_with("linux") => {
448            surface.append_terminal_line(&alloc::format!("[*] Linux ABI Subsystem: {}", cmd));
449            if cmd == "linux status" {
450                surface.append_terminal_line("Linux ABI: 6.8.0-hydra (x86_64) | glibc & musl ready");
451                surface.append_terminal_line("Wayland Display: /run/user/1000/wayland-0 (Active)");
452                surface.append_terminal_line("Dynamic Linker: /lib64/ld-hydra.so | auxv online");
453                surface.append_terminal_line("Input Routing: wl_seat (pointer & keyboard streaming)");
454            } else if cmd.contains("run") {
455                surface.append_terminal_line("[+] Executed Linux ELF binary in Ring 3 userspace.");
456                surface.append_terminal_line("[+] Wayland buffer attached and committed to display.");
457            }
458        }
459        other => {
460            surface.append_terminal_line(&alloc::format!("konsole: command not found: '{}'", other));
461        }
462    }
