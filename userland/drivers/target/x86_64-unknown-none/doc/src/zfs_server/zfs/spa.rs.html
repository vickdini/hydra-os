1//! # Storage Pool Allocator (SPA)
2//!
3//! ## Subsystem Role & Architecture
4//! The Storage Pool Allocator (SPA) is the bottom layer of Hydra's OpenZFS implementation.
5//! It abstracts physical storage into a virtual device ([`Vdev`]), manages block allocation
6//! via a persistent space bitmap, maintains an in-memory physical frame cache ([`BlockCache`]),
7//! and controls the uberblock ring for atomic commit semantics.
8//!
9//! ## Crash Consistency & Transaction Groups
10//! Blocks freed in transaction group N are still referenced by uberblock N-1, so they are
11//! only returned to the allocator after uberblock N is on disk, and only reach the on-disk
12//! bitmap in the commit of txg N+1. A crash can therefore leak blocks allocated by an
13//! unfinished commit, but never hands out a block the last valid uberblock still uses.
14//!
15//! ## Key Types & Interfaces
16//! - [`Vdev`]: The disk under the pool, driven by the Ring 3 disk driver (`ata-driver`), or
17//!   a RAM disk.
18//! - [`BlockCache`]: LRU physical frame cache avoiding redundant device read cycles.
19//! - [`Spa`]: Storage Pool Allocator controlling allocation cursors, bitmaps, and labels.
20//! - [`DiskContents`]: Detection state for disk inspection prior to pool import.
21//!
22//! ## Concurrency & Hardware Semantics
23//! The kernel has no disk driver of its own. Requests to the disk driver are calls to its
24//! service ([`crate::ipc::service::call`]), which run the driver in the caller's place, so
25//! they may be made with the file system's locks held. Allocations and frees are staged
26//! in transaction groups and committed atomically.
27

39/// Logical block size in bytes for SPA allocations (4 KiB).
40pub const BLOCK: u64 = ZFS_BLOCK_SIZE as u64;
41/// Disk sector size the pool is laid out in.
42const SECTOR: u64 = 512;
43/// Largest number of consecutive blocks gathered into one disk request (one disk driver
44/// window).
45const MAX_RUN: usize = crate::ipc::service::WINDOW_PAGES as usize;
46/// Size of the RAM-backed pool used when no ATA disk is present.
47const RAM_POOL_BYTES: u64 = 64 * 1024 * 1024;
48/// Default cap on cached blocks (128 MiB of frames).
49const CACHE_BLOCKS: usize = 32 * 1024;

51/// Returns the 4 KiB contents of a frame through the identity map.
52pub fn frame_bytes(frame: PhysFrame<Size4KiB>) -> &'static mut [u8; ZFS_BLOCK_SIZE] {
53    unsafe { &mut *(frame.start_address().as_u64() as *mut [u8; ZFS_BLOCK_SIZE]) }

84/// The disk driver's service window (starting the driver if it is not running).
85fn driver_window() -> Result<&'static mut [u8], &'static str> {
86    if let Some(w) = service::window(ENDPOINT_BLOCK_DEV) {
87        return Ok(w);
88    }
89    crate::task::supervisor::revive(ENDPOINT_BLOCK_DEV);
90    service::window(ENDPOINT_BLOCK_DEV).ok_or("ZFS: the disk driver is not running")
91}

93/// Sends a request to the disk driver; returns its reply or an error message.
94fn driver_request(label: u32, args: [u64; 4], what: &'static str) -> Result<crate::ipc::Message, &'static str> {
95    match service::call(ENDPOINT_BLOCK_DEV, crate::ipc::Message::with_args(label, args), service::CALL_TIMEOUT_TICKS) {
96        Ok(r) if r.label == MSG_BLOCK_REPLY && (r.args[0] as i64) >= 0 => Ok(r),
97        Ok(_) => Err(what),
98        Err(_) => Err("ZFS: the disk driver did not answer"),
99    }
100}

111impl Vdev {
112    /// Opens the disk through the disk driver (started at the hardware takeover), or a RAM
113    /// disk if there is none.
114    pub fn open() -> Self {
115        if service::is_online(ENDPOINT_BLOCK_DEV) {
116            match Self::query_driver() {
117                Ok(blocks) => return Self { backend: Backend::Driver, total_blocks: blocks },
118                Err(e) => crate::println!("[!] {}", e),
119            }
120        }
121        let pages = (RAM_POOL_BYTES / BLOCK) as usize;
122        let base = FRAME_ALLOCATOR
123            .lock()
124            .allocate_contiguous_frames(pages)
125            .map(|f| f.start_address().as_u64() as *mut u8)
126            .unwrap_or(core::ptr::null_mut());
127        if !base.is_null() {
128            unsafe { core::ptr::write_bytes(base, 0, RAM_POOL_BYTES as usize) };
129        }
130        Self {
131            backend: Backend::Ram(base),
132            total_blocks: if base.is_null() { 0 } else { RAM_POOL_BYTES / BLOCK },
133        }
134    }
135
136    /// Asks the disk driver about its disk; returns its size in blocks.
137    fn query_driver() -> Result<u64, &'static str> {
138        let reply = driver_request(MSG_BLOCK_GET_SIZE, [0; 4], "ZFS: the disk driver cannot size the disk")?;
139        let window = driver_window()?;
140        let model = core::str::from_utf8(&window[..40]).unwrap_or("").trim_end_matches(|c: char| c == ' ' || c == '\0');
141        *DISK_INFO.lock() = Some(DiskInfo {
142            model: String::from(model),
143            sectors: reply.args[1] * (BLOCK / SECTOR),
144            lba48: reply.args[2] != 0,
145            dma: reply.args[3] != 0,
146        });
147        Ok(reply.args[1])
148    }
149
150    /// Returns true if this virtual device is a disk rather than RAM.
151    pub fn is_hardware_backed(&self) -> bool {
152        matches!(self.backend, Backend::Driver)
153    }
154
155    /// Reads `count` consecutive blocks starting at `block` into `buf`.
156    pub fn read(&self, block: u64, buf: &mut [u8]) -> Result<(), &'static str> {
157        let count = buf.len() / ZFS_BLOCK_SIZE;
158        if block + count as u64 > self.total_blocks {
159            return Err("ZFS: block read past end of vdev");
160        }
161        match &self.backend {
162            Backend::Ram(base) => unsafe {
163                core::ptr::copy_nonoverlapping(base.add((block * BLOCK) as usize), buf.as_mut_ptr(), buf.len());
164                Ok(())
165            },
166            Backend::Driver => {
167                let run = service::WINDOW_SIZE;
168                for (i, chunk) in buf.chunks_mut(run).enumerate() {
169                    let n = (chunk.len() / ZFS_BLOCK_SIZE) as u64;
170                    let at = block + (i * run / ZFS_BLOCK_SIZE) as u64;
171                    driver_request(MSG_BLOCK_READ, [at, n, 0, 0], "ZFS: disk read failed")?;
172                    chunk.copy_from_slice(&driver_window()?[..chunk.len()]);
173                }
174                Ok(())
175            }
176        }
177    }
178
179    /// Writes `buf` (a whole number of blocks) starting at `block`.
180    pub fn write(&mut self, block: u64, buf: &[u8]) -> Result<(), &'static str> {
181        let count = buf.len() / ZFS_BLOCK_SIZE;
182        if block + count as u64 > self.total_blocks {
183            return Err("ZFS: block write past end of vdev");
184        }
185        match &self.backend {
186            Backend::Ram(base) => unsafe {
187                core::ptr::copy_nonoverlapping(buf.as_ptr(), base.add((block * BLOCK) as usize), buf.len());
188                Ok(())
189            },
190            Backend::Driver => {
191                let run = service::WINDOW_SIZE;
192                for (i, chunk) in buf.chunks(run).enumerate() {
193                    let n = (chunk.len() / ZFS_BLOCK_SIZE) as u64;
194                    let at = block + (i * run / ZFS_BLOCK_SIZE) as u64;
195                    driver_window()?[..chunk.len()].copy_from_slice(chunk);
196                    driver_request(MSG_BLOCK_WRITE, [at, n, 0, 0], "ZFS: disk write failed")?;
197                }
198                Ok(())
199            }
200        }
201    }
202
203    /// Flushes the disk's write cache to non-volatile storage.
204    pub fn flush(&mut self) -> Result<(), &'static str> {
205        match &self.backend {
206            Backend::Driver => driver_request(MSG_BLOCK_FLUSH, [0; 4], "ZFS: disk cache flush failed").map(|_| ()),
207            Backend::Ram(_) => Ok(()),
208        }
209    }

217/// Bounded cache of on-disk blocks, stored in physical frames.
218pub struct BlockCache {
219    map: BTreeMap<u64, CacheEntry>,
220    clock: u64,
221    capacity: usize,
222    /// Number of successful block cache lookups.
223    pub hits: u64,
224    /// Number of cache lookup misses requiring disk reads.
225    pub misses: u64,
226}
227
228impl BlockCache {
229    fn new(capacity: usize) -> Self {
230        Self {
231            map: BTreeMap::new(),
232            clock: 0,
233            capacity,
234            hits: 0,
235            misses: 0,
236        }
237    }
238
239    fn get(&mut self, block: u64) -> Option<PhysFrame<Size4KiB>> {
240        self.clock += 1;
241        let clock = self.clock;
242        self.map.get_mut(&block).map(|e| {
243            e.last_use = clock;
244            e.frame
245        })
246    }
247
248    /// Stores a copy of `data` for `block`.
249    fn put(&mut self, block: u64, data: &[u8; ZFS_BLOCK_SIZE]) {
250        self.clock += 1;
251        if let Some(e) = self.map.get_mut(&block) {
252            frame_bytes(e.frame).copy_from_slice(data);
253            e.last_use = self.clock;
254            return;
255        }
256        if self.map.len() >= self.capacity {
257            self.evict();
258        }
259        if let Some(frame) = alloc_user_frame() {
260            frame_bytes(frame).copy_from_slice(data);
261            self.map.insert(block, CacheEntry { frame, last_use: self.clock });
262        }
263    }
264
265    /// Drops the least recently used eighth of the cache.
266    fn evict(&mut self) {
267        let mut ages: Vec<u64> = self.map.values().map(|e| e.last_use).collect();
268        ages.sort_unstable();
269        let cutoff = ages[ages.len() / 8];
270        let victims: Vec<u64> = self.map.iter().filter(|(_, e)| e.last_use <= cutoff).map(|(&k, _)| k).collect();
271        for k in victims {
272            if let Some(e) = self.map.remove(&k) {
273                frame_ref_dec(e.frame);
274            }
275        }
276    }
277
278    fn invalidate(&mut self, block: u64) {
279        if let Some(e) = self.map.remove(&block) {
280            frame_ref_dec(e.frame);
281        }
282    }
283
284    /// Returns the number of blocks currently held in the block cache.
285    pub fn len(&self) -> usize {
286        self.map.len()
287    }
288
289    /// Discards every cached block (forces later reads to come from the device).
290    pub fn clear(&mut self) {
291        for (_, e) in core::mem::take(&mut self.map) {
292            frame_ref_dec(e.frame);
293        }
294    }
295}

308/// Storage Pool Allocator.

311    pub vdev: Vdev,
312    /// On-disk pool configuration label.
313    pub label: VdevLabel,
314    /// In-memory cache of clean and dirty block frames.
315    pub cache: BlockCache,

326impl Spa {
327    fn empty(vdev: Vdev, label: VdevLabel) -> Self {
328        let words = ((label.total_blocks + 63) / 64) as usize;
329        Self {
330            vdev,
331            label,
332            cache: BlockCache::new(CACHE_BLOCKS),
333            bitmap: alloc::vec![0u64; words],
334            dirty_bitmap_blocks: BTreeMap::new(),
335            pending_free: Vec::new(),
336            alloc_cursor: label.data_start,
337            allocated_blocks: 0,
338            pool_name: label.pool_name,
339        }
340    }

342    /// Lays out a brand-new pool on `vdev`. The caller commits the first uberblock.
343    pub fn create(mut vdev: Vdev, name: &str) -> Result<Self, &'static str> {
344        let total = vdev.total_blocks;
345        if total < 256 {
346            return Err("ZFS: device too small for a pool");
347        }
348        let bitmap_blocks = (total + BLOCK * 8 - 1) / (BLOCK * 8);
349        let mut pool_name = [0u8; 32];
350        let n = name.len().min(31);
351        pool_name[..n].copy_from_slice(&name.as_bytes()[..n]);
352        let mut label = VdevLabel {
353            magic: super::types::VDEV_LABEL_MAGIC,
354            version: super::types::POOL_VERSION,
355            pool_guid: unsafe { core::arch::x86_64::_rdtsc() } ^ 0x4859_4452_4150_4f4f,
356            total_blocks: total,
357            bitmap_start: 1 + UBERBLOCK_RING,
358            bitmap_blocks,
359            ub_start: 1,
360            ub_count: UBERBLOCK_RING,
361            data_start: 1 + UBERBLOCK_RING + bitmap_blocks,
362            pool_name,
363            checksum: Fletcher4::default(),
364        };
365        label.seal();
366
367        // Write the label and clear the uberblock ring and bitmap region
368        let mut block = [0u8; ZFS_BLOCK_SIZE];
369        unsafe {
370            core::ptr::copy_nonoverlapping(&label as *const VdevLabel as *const u8, block.as_mut_ptr(), core::mem::size_of::<VdevLabel>());
371        }
372        vdev.write(0, &block)?;
373        let zeros = alloc::vec![0u8; ZFS_BLOCK_SIZE * MAX_RUN];
374        let mut b = 1;
375        while b < label.data_start {
376            let n = ((label.data_start - b) as usize).min(MAX_RUN);
377            vdev.write(b, &zeros[..n * ZFS_BLOCK_SIZE])?;
378            b += n as u64;
379        }
380
381        let mut spa = Self::empty(vdev, label);
382        for blk in 0..label.data_start {
383            spa.mark(blk, true);
384        }
385        spa.allocated_blocks = label.data_start;
386        spa.fence_tail();
387        Ok(spa)
388    }

390    /// Probes block 0 of a device to determine its contents: nothing (all zeros), a pool
391    /// label, or foreign data. Read errors are retried; returns `None` if device read fails.
392    pub fn probe(vdev: &Vdev) -> Option<DiskContents> {
393        let mut block = [0u8; ZFS_BLOCK_SIZE];
394        for _ in 0..3 {
395            if vdev.read(0, &mut block).is_ok() {
396                if block.iter().all(|&b| b == 0) {
397                    return Some(DiskContents::Blank);
398                }
399                let label: VdevLabel = unsafe { core::ptr::read_unaligned(block.as_ptr() as *const VdevLabel) };
400                return Some(if label.magic == super::types::VDEV_LABEL_MAGIC { DiskContents::Pool } else { DiskContents::Other });
401            }
402        }
403        None
404    }

406    /// Opens an existing pool, returning it with its newest valid uberblock, or hands the
407    /// device back if it holds no valid pool.
408    ///
409    /// # Arguments
410    /// * `vdev` - Virtual block device to inspect and mount.
411    ///
412    /// # Returns
413    /// Tuple of `(Spa, Uberblock)` on success, or `Err(vdev)` on failure.
414    pub fn open(vdev: Vdev) -> Result<(Self, Uberblock), Vdev> {
415        let mut block = [0u8; ZFS_BLOCK_SIZE];
416        if vdev.read(0, &mut block).is_err() {
417            return Err(vdev);
418        }
419        let label: VdevLabel = unsafe { core::ptr::read_unaligned(block.as_ptr() as *const VdevLabel) };
420        if !label.is_valid() || label.total_blocks > vdev.total_blocks {
421            return Err(vdev);
422        }
423
424        let mut best: Option<Uberblock> = None;
425        for slot in 0..label.ub_count {
426            if vdev.read(label.ub_start + slot, &mut block).is_err() {
427                continue;
428            }
429            let ub: Uberblock = unsafe { core::ptr::read_unaligned(block.as_ptr() as *const Uberblock) };
430            if ub.is_valid() && best.map_or(true, |b| ub.txg > b.txg) {
431                best = Some(ub);
432            }
433        }
434        let ub = match best {
435            Some(ub) => ub,
436            None => return Err(vdev),
437        };
438
439        let mut spa = Self::empty(vdev, label);
440        let mut buf = alloc::vec![0u8; ZFS_BLOCK_SIZE * MAX_RUN];
441        let mut b = 0u64;
442        while b < label.bitmap_blocks {
443            let n = ((label.bitmap_blocks - b) as usize).min(MAX_RUN);
444            if spa.vdev.read(label.bitmap_start + b, &mut buf[..n * ZFS_BLOCK_SIZE]).is_err() {
445                return Err(spa.vdev);
446            }
447            for (i, word) in buf[..n * ZFS_BLOCK_SIZE].chunks_exact(8).enumerate() {
448                let idx = (b as usize * ZFS_BLOCK_SIZE / 8) + i;
449                if idx < spa.bitmap.len() {
450                    spa.bitmap[idx] = u64::from_le_bytes(word.try_into().unwrap());
451                }
452            }
453            b += n as u64;
454        }
455        let tail = spa.bitmap.len() as u64 * 64 - label.total_blocks;
456        spa.fence_tail();
457        spa.allocated_blocks = spa.bitmap.iter().map(|w| w.count_ones() as u64).sum::<u64>() - tail;
458        Ok((spa, ub))
459    }

461    fn mark(&mut self, block: u64, used: bool) {
462        let (w, bit) = ((block / 64) as usize, block % 64);
463        if used {
464            self.bitmap[w] |= 1 << bit;
465        } else {
466            self.bitmap[w] &= !(1 << bit);
467        }
468        self.dirty_bitmap_blocks.insert(block / (BLOCK * 8), ());
469    }
470
471    fn is_used(&self, block: u64) -> bool {
472        self.bitmap[(block / 64) as usize] & (1 << (block % 64)) != 0
473    }

475    /// Allocates one data block, scanning the bitmap a word at a time from the cursor.
476    pub fn alloc(&mut self) -> Result<u64, &'static str> {
477        let words = self.bitmap.len();
478        let start_word = (self.alloc_cursor.max(self.label.data_start) / 64) as usize % words;
479        for i in 0..=words {
480            let w = (start_word + i) % words;
481            let word = self.bitmap[w];
482            if word != u64::MAX {
483                let b = w as u64 * 64 + (!word).trailing_zeros() as u64;
484                self.mark(b, true);
485                self.allocated_blocks += 1;
486                self.alloc_cursor = b + 1;
487                return Ok(b);
488            }
489        }
490        Err("ZFS: pool is out of space")
491    }

493    /// Marks the bits past the end of the device in the last bitmap word as used so the
494    /// allocator never returns them (in memory only; not counted as allocated).
495    fn fence_tail(&mut self) {
496        let total = self.label.total_blocks;
497        if total % 64 != 0 {
498            let last = self.bitmap.len() - 1;
499            self.bitmap[last] |= !0u64 << (total % 64);
500        }
501    }

503    /// Schedules a block to be freed once the current transaction group is on disk.
504    pub fn free_deferred(&mut self, block: u64) {
505        if block >= self.label.data_start {
506            self.pending_free.push(block);
507        }
508    }

510    /// Reads a block, verifying it against its block pointer, via the cache.
511    pub fn read_bp(&mut self, bp: &BlockPointer) -> Result<PhysFrame<Size4KiB>, &'static str> {
512        if let Some(frame) = self.cache.get(bp.block) {
513            self.cache.hits += 1;
514            return Ok(frame);
515        }
516        self.cache.misses += 1;
517        let mut data = [0u8; ZFS_BLOCK_SIZE];
518        self.vdev.read(bp.block, &mut data)?;
519        if Fletcher4::compute(&data[..bp.lsize as usize]) != bp.checksum {
520            return Err("ZFS Checksum verification failed (silent data corruption detected)");
521        }
522        self.cache.put(bp.block, &data);
523        self.cache.get(bp.block).ok_or("ZFS: out of memory for block cache")
524    }

526    /// Writes a batch of freshly allocated blocks, coalescing consecutive block numbers
527    /// into multi-block device writes, and caches them.
528    pub fn write_blocks(&mut self, mut blocks: Vec<(u64, PhysFrame<Size4KiB>)>) -> Result<(), &'static str> {
529        blocks.sort_unstable_by_key(|(b, _)| *b);
530        let mut run: Vec<u8> = Vec::with_capacity(ZFS_BLOCK_SIZE * MAX_RUN);
531        let mut i = 0;
532        while i < blocks.len() {
533            let start = blocks[i].0;
534            run.clear();
535            let mut n = 0;
536            while i < blocks.len() && blocks[i].0 == start + n as u64 && n < MAX_RUN {
537                run.extend_from_slice(frame_bytes(blocks[i].1));
538                self.cache.put(blocks[i].0, frame_bytes(blocks[i].1));
539                i += 1;
540                n += 1;
541            }
542            self.vdev.write(start, &run)?;
543        }
544        Ok(())
545    }

547    /// Finishes a transaction group: persists dirty bitmap blocks, writes the uberblock,
548    /// then releases blocks freed during the group.
549    pub fn commit(&mut self, ub: &mut Uberblock) -> Result<(), &'static str> {
550        // Blocks stay dirty until written, so a failed commit is redone by the next one
551        let dirty: Vec<u64> = self.dirty_bitmap_blocks.keys().copied().collect();
552        for idx in dirty {
553            let mut block = [0u8; ZFS_BLOCK_SIZE];
554            let first_word = idx as usize * ZFS_BLOCK_SIZE / 8;
555            for (i, chunk) in block.chunks_exact_mut(8).enumerate() {
556                if let Some(w) = self.bitmap.get(first_word + i) {
557                    chunk.copy_from_slice(&w.to_le_bytes());
558                }
559            }
560            self.vdev.write(self.label.bitmap_start + idx, &block)?;
561            self.dirty_bitmap_blocks.remove(&idx);
562        }
563        self.vdev.flush()?;
564
565        ub.allocated_blocks = self.allocated_blocks;
566        ub.seal();
567        let mut block = [0u8; ZFS_BLOCK_SIZE];
568        unsafe {
569            core::ptr::copy_nonoverlapping(ub as *const Uberblock as *const u8, block.as_mut_ptr(), core::mem::size_of::<Uberblock>());
570        }
571        self.vdev.write(self.label.ub_start + ub.txg % self.label.ub_count, &block)?;
572        self.vdev.flush()?;
573
574        // The new uberblock is durable: blocks freed in this group may now be reused
575        let freed = core::mem::take(&mut self.pending_free);
576        for b in freed {
577            if b < self.label.total_blocks && self.is_used(b) {
578                self.mark(b, false);
579                self.allocated_blocks -= 1;
580                self.cache.invalidate(b);
581            }
582        }
583        Ok(())
584    }
585
586    /// Returns the pool name as a UTF-8 string slice.
587    pub fn pool_name_str(&self) -> &str {
588        let len = self.pool_name.iter().position(|&b| b == 0).unwrap_or(self.pool_name.len());
589        core::str::from_utf8(&self.pool_name[..len]).unwrap_or("hydrapool")
590    }
591
592    /// Returns the total capacity of the pool in bytes.
593    pub fn total_capacity_bytes(&self) -> u64 {
594        self.label.total_blocks * BLOCK
595    }
596
597    /// Returns the total number of bytes currently allocated in the pool.
598    pub fn allocated_bytes(&self) -> u64 {
599        self.allocated_blocks * BLOCK
600    }
601
602    /// Returns the remaining free space in the pool in bytes.
603    pub fn free_bytes(&self) -> u64 {
604        self.total_capacity_bytes().saturating_sub(self.allocated_bytes())
605    }
606
607    /// Returns true if the pool's virtual device is backed by physical hardware.
608    pub fn is_hardware_backed(&self) -> bool {
609        self.vdev.is_hardware_backed()
610    }
