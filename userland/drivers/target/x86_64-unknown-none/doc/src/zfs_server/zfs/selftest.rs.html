1//! # ZFS Self-Test
2//!
3//! `zfs selftest` builds a test tree under `/zfstest`, verifies it, commits, drops every
4//! cache and verifies again from disk. `zfs verify` re-checks the tree after a reboot.

14const BIG_FILE_BYTES: usize = 5 * 1024 * 1024 + 1234;
15const MANY_FILES: usize = 3000;

17fn pattern(i: usize) -> u8 {
18    (i.wrapping_mul(2_654_435_761) >> 13) as u8
19}

21fn check(name: &str, ok: bool, failures: &mut u32) {
22    println!("[zfstest] {:<52} {}", name, if ok { "PASS" } else { "FAIL" });
23    if !ok {
24        *failures += 1;
25    }
26}

28fn path(s: &mut ZfsServer, p: &str) -> Result<u64, i64> {
29    let root = s.root_obj();
30    resolve_at(s, root, p.as_bytes(), true)
31}

33/// Removes the directory tree `dir/name` recursively.
34fn remove_tree(s: &mut ZfsServer, dir: u64, name: &[u8]) -> Result<(), i64> {
35    let ino = s.lookup(dir, name)?;
36    if s.getattr(ino)?.is_dir() {
37        loop {
38            let batch = s.readdir(ino, 2, 64)?;
39            if batch.is_empty() {
40                break;
41            }
42            for e in batch {
43                remove_tree(s, ino, &e.name)?;
44            }
45        }
46        s.rmdir(dir, name)
47    } else {
48        s.unlink(dir, name)
49    }
50}

52fn build(s: &mut ZfsServer) -> Result<(), i64> {
53    let root = s.root_obj();
54    if s.lookup(root, b"zfstest").is_ok() {
55        println!("[zfstest] removing /zfstest left by an earlier run");
56        remove_tree(s, root, b"zfstest")?;
57    }
58    let top = s.mkdir(root, b"zfstest", 0o755, 0, 0)?;
59
60    // Large file spanning several indirect levels
61    let big = s.create(top, b"big.bin", 0o644, 1000, 1000)?;
62    let chunk: Vec<u8> = (0..65536).map(pattern).collect();
63    let mut off = 0;
64    while off < BIG_FILE_BYTES {
65        let n = (BIG_FILE_BYTES - off).min(chunk.len());
66        let data: Vec<u8> = (off..off + n).map(pattern).collect();
67        s.write(big, off as u64, &data)?;
68        off += n;
69    }
70
71    // Sparse file: data far past the end leaves a hole
72    let sparse = s.create(top, b"sparse.bin", 0o600, 0, 0)?;
73    s.write(sparse, 300 * 1024 * 1024, b"tail")?;
74
75    // Many directory entries with long names
76    let many = s.mkdir(top, b"many", 0o755, 0, 0)?;
77    for i in 0..MANY_FILES {
78        let name = format!("entry-{:05}-{}", i, "x".repeat(i % 200));
79        let ino = s.create(many, name.as_bytes(), 0o644, 0, 0)?;
80        s.write(ino, 0, format!("file {}", i).as_bytes())?;
81    }
82    // Remove every third entry to exercise tombstones and slot reuse
83    for i in (0..MANY_FILES).step_by(3) {
84        let name = format!("entry-{:05}-{}", i, "x".repeat(i % 200));
85        s.unlink(many, name.as_bytes())?;
86    }
87
88    // Symlinks (inline and long), hard links, rename, truncate
89    s.symlink(top, b"link-short", b"big.bin", 0, 0)?;
90    let long_target = format!("/zfstest{}/big.bin", "/sub/..".repeat(40));
91    s.symlink(top, b"link-long", long_target.as_bytes(), 0, 0)?;
92    s.link(big, top, b"big-hardlink")?;
93    let tmp = s.create(top, b"rename-me", 0o644, 0, 0)?;
94    s.write(tmp, 0, b"renamed contents")?;
95    let sub = s.mkdir(top, b"sub", 0o750, 0, 0)?;
96    s.rename(top, b"rename-me", sub, b"renamed")?;
97    let trunc = s.create(top, b"truncated", 0o644, 0, 0)?;
98    s.write(trunc, 0, &[0xAB; 10_000])?;
99    s.setattr(trunc, SetAttr { size: Some(5000), ..SetAttr::default() })?;
100    s.setattr(trunc, SetAttr { size: Some(8192), ..SetAttr::default() })?;
101    Ok(())
102}

104fn verify(s: &mut ZfsServer, failures: &mut u32) -> Result<(), i64> {
105    let big = path(s, "/zfstest/big.bin")?;
106    let attr = s.getattr(big)?;
107    check("big.bin size", attr.size == BIG_FILE_BYTES as u64, failures);
108    check("big.bin nlink == 2 (hard link)", attr.nlink == 2, failures);
109    let mut buf = alloc::vec![0u8; 65536];
110    let mut ok = true;
111    let mut off = 0;
112    while off < BIG_FILE_BYTES {
113        let n = s.read(big, off as u64, &mut buf)?;
114        ok &= n > 0 && buf[..n].iter().enumerate().all(|(i, &b)| b == pattern(off + i));
115        off += n.max(1);
116    }
117    check("big.bin contents (multi-level block tree)", ok, failures);
118
119    let sparse = path(s, "/zfstest/sparse.bin")?;
120    let mut hole = [1u8; 4096];
121    s.read(sparse, 100 * 1024 * 1024, &mut hole)?;
122    let mut tail = [0u8; 4];
123    s.read(sparse, 300 * 1024 * 1024, &mut tail)?;
124    check("sparse.bin hole reads zero, tail intact", hole.iter().all(|&b| b == 0) && &tail == b"tail", failures);
125
126    let many = path(s, "/zfstest/many")?;
127    let mut names = 0;
128    let mut cookie = 0;
129    let mut content_ok = true;
130    loop {
131        let batch = s.readdir(many, cookie, 128)?;
132        if batch.is_empty() {
133            break;
134        }
135        for e in batch {
136            cookie = e.next_cookie;
137            if e.name == b"." || e.name == b".." {
138                continue;
139            }
140            names += 1;
141            let idx: usize = core::str::from_utf8(&e.name[6..11]).ok().and_then(|n| n.parse().ok()).unwrap_or(usize::MAX);
142            let mut b = [0u8; 16];
143            let n = s.read(e.ino, 0, &mut b)?;
144            content_ok &= idx % 3 != 0 && b[..n] == *format!("file {}", idx).as_bytes();
145        }
146    }
147    check("directory with 3000 entries, 1000 removed", names == MANY_FILES - MANY_FILES / 3, failures);
148    check("directory entries resolve to the right files", content_ok, failures);
149
150    let short = path(s, "/zfstest/link-short")?;
151    check("short symlink resolves", short == big, failures);
152    let long = path(s, "/zfstest/link-long")?;
153    check("long symlink (data blocks) resolves", long == big, failures);
154    let root = s.root_obj();
155    let link_ino = resolve_at(s, root, b"/zfstest/link-short", false)?;
156    check("symlink lstat is S_IFLNK", s.getattr(link_ino)?.mode & S_IFMT == S_IFLNK, failures);
157
158    let renamed = path(s, "/zfstest/sub/renamed")?;
159    let mut r = [0u8; 32];
160    let n = s.read(renamed, 0, &mut r)?;
161    check("rename across directories", &r[..n] == b"renamed contents", failures);
162    check("old name gone after rename", path(s, "/zfstest/rename-me") == Err(zpl::ENOENT), failures);
163    let sub = path(s, "/zfstest/sub")?;
164    check("subdir mode 0750", s.getattr(sub)?.mode & 0o7777 == 0o750, failures);
165    let up = path(s, "/zfstest/sub/..")?;
166    let top = path(s, "/zfstest")?;
167    check(".. of subdir is parent", up == top, failures);
168
169    let trunc = path(s, "/zfstest/truncated")?;
170    let mut t = alloc::vec![0u8; 8192];
171    let n = s.read(trunc, 0, &mut t)?;
172    check(
173        "truncate down then up zero-fills",
174        n == 8192 && t[..5000].iter().all(|&b| b == 0xAB) && t[5000..].iter().all(|&b| b == 0),
175        failures,
176    );
177    Ok(())
178}

180fn report(failures: u32) {
181    println!("[zfstest] {} ({} failure{})", if failures == 0 { "ALL PASSED" } else { "FAILED" }, failures, if failures == 1 { "" } else { "s" });
182}

184/// Creates the test tree and verifies it before and after a commit with cold caches.
185pub fn run_selftest() {
186    let mut failures = 0;
187    let res = with_server(|s| {
188        build(s)?;
189        verify(s, &mut failures)?;
190        let txg = s.sync()?;
191        println!("[zfstest] committed TXG #{}; dropping caches and re-reading from disk", txg);
192        s.dmu.drop_caches();
193        s.zap.clear();
194        verify(s, &mut failures)
195    });
196    if let Err(e) = res {
197        println!("[zfstest] aborted with errno {}", e);
198        failures += 1;
199    }
200    report(failures);
201}

203/// Re-verifies a test tree created by an earlier boot.
204pub fn run_verify() {
205    let mut failures = 0;
206    if let Err(e) = with_server(|s| verify(s, &mut failures)) {
207        println!("[zfstest] aborted with errno {}", e);
208        failures += 1;
209    }
210    report(failures);
211}

213/// `zfs fsck <dir> [repair]`: walks the namespace below `dir` and reports directory
214/// entries whose object cannot be read (left behind when a crash loses blocks). With
215/// `repair` those entries are removed, so the names can be created again.
216pub fn run_fsck(root_path: &str, repair: bool) {
217    let r = with_server(|s| {
218        let root = path(s, root_path)?;
219        let (mut dirs, mut entries, mut dangling, mut bad_dirs) = (0u64, 0u64, 0u64, 0u64);
220        // (directory, its path, its parent and name there: unreadable ones are unlinked)
221        let mut stack: Vec<(u64, alloc::string::String, Option<(u64, Vec<u8>)>)> =
222            alloc::vec![(root, alloc::string::String::from(root_path.trim_end_matches('/')), None)];
223        while let Some((dir, dpath, link)) = stack.pop() {
224            dirs += 1;
225            if dirs % 2000 == 0 {
226                println!("[fsck] ... {} directories, {} entries, {} dangling (at {})", dirs, entries, dangling, dpath);
227            }
228            let mut cookie = 2; // after "." and ".."
229            let mut doomed: Vec<Vec<u8>> = Vec::new();
230            loop {
231                let batch = match s.readdir(dir, cookie, 256) {
232                    Ok(b) => b,
233                    Err(e) => {
234                        bad_dirs += 1;
235                        println!("[fsck] unreadable directory {} (errno {})", dpath, -e);
236                        if repair {
237                            if let Some((parent, name)) = &link {
238                                s.zap.remove(&mut s.dmu, *parent, name).map_err(|_| -5i64)?;
239                            }
240                        }
241                        break;
242                    }
243                };
244                if batch.is_empty() {
245                    break;
246                }
247                if batch.last().map_or(true, |e| e.next_cookie <= cookie) {
248                    bad_dirs += 1;
249                    println!("[fsck] directory {} does not advance past position {}", dpath, cookie);
250                    break;
251                }
252                for e in &batch {
253                    cookie = e.next_cookie;
254                    entries += 1;
255                    let name = alloc::string::String::from_utf8_lossy(&e.name).into_owned();
256                    match s.getattr(e.ino) {
257                        Ok(a) if a.is_dir() => stack.push((e.ino, format!("{}/{}", dpath, name), Some((dir, e.name.clone())))),
258                        Ok(_) => {}
259                        Err(err) => {
260                            dangling += 1;
261                            if dangling <= 200 {
262                                println!("[fsck] dangling entry {}/{} -> object {} (errno {})", dpath, name, e.ino, -err);
263                            }
264                            doomed.push(e.name.clone());
265                        }
266                    }
267                }
268            }
269            if repair {
270                for name in doomed {
271                    s.zap.remove(&mut s.dmu, dir, &name).map_err(|_| -5i64)?;
272                }
273            }
274        }
275        if repair && (dangling > 0 || bad_dirs > 0) {
276            s.sync()?;
277        }
278        println!(
279            "[fsck] {}: {} directories, {} entries, {} dangling{}, {} unreadable directories",
280            root_path,
281            dirs,
282            entries,
283            dangling,
284            if repair { " (removed)" } else { "" },
285            bad_dirs
286        );
287        Ok(())
288    });
289    if let Err(e) = r {
290        println!("[fsck] failed: errno {}", -e);
291    }
