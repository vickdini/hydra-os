1//! # File System Protocol: the kernel's requests to the ZFS file server
2//!
3//! The OpenZFS implementation runs as a Ring 3 program (`userland/drivers/zfs-server`); the
4//! kernel's file layer asks it for ZPL operations (POSIX-style, by inode) over
5//! `ENDPOINT_ZFS_SERVER`. This module is shared by both sides
6//! (the server includes it with `#[path]`): the operation codes, the types they carry, and
7//! their encoding in the service window.
8//!
9//! ## Wire format
10//!
11//! A request is a message whose label is the operation; its parameters are encoded in the
12//! window from offset 0 (`Writer`: little-endian `u64`s and length-prefixed byte strings).
13//! The reply carries `args = [status (value or -errno), value, reply bytes, events]`: reply
14//! data is in the window from offset 0, and the inotify events the operation caused are at
15//! `EVENT_OFFSET` (the kernel delivers them to its watchers).
16

121/// Attributes of an inode (the fields of Linux `struct stat`).
122#[derive(Debug, Clone, Copy, Default)]
123pub struct Attr {
124    /// Inode / object identifier.
125    pub ino: u64,
126    /// File mode and permission bits (e.g. `S_IFREG | 0o644`).
127    pub mode: u32,
128    /// Number of hard links pointing to this inode.
129    pub nlink: u32,
130    /// Owner user ID.
131    pub uid: u32,
132    /// Owner group ID.
133    pub gid: u32,
134    /// Device ID for character / block special files.
135    pub rdev: u64,
136    /// Logical size of the file in bytes.
137    pub size: u64,
138    /// Allocated size in 512-byte units.
139    pub blocks: u64,
140    /// Last access timestamp as (seconds, nanoseconds).
141    pub atime: (i64, u32),
142    /// Last modification timestamp as (seconds, nanoseconds).
143    pub mtime: (i64, u32),
144    /// Status change timestamp as (seconds, nanoseconds).
145    pub ctime: (i64, u32),
146    /// Inode generation counter for NFS / stale handle detection.
147    pub generation: u64,
148    /// Transaction group in which this inode was created.
149    pub birth_txg: u64,
150}
151
152impl Attr {
153    /// Determines the dnode type from the mode bits.
154    pub fn kind(&self) -> DnodeType {
155        DnodeType::from_mode(self.mode)
156    }
157
158    /// Returns true if this inode represents a directory.
159    pub fn is_dir(&self) -> bool {
160        self.mode & S_IFMT == S_IFDIR
161    }
162}

235/// Encodes parameters or results into the window.
236pub struct Writer<'a> {
237    buf: &'a mut [u8],
238    pos: usize,
239}
240
241impl<'a> Writer<'a> {
242    /// Writes from the start of `buf`.
243    pub fn new(buf: &'a mut [u8]) -> Self {
244        Self { buf, pos: 0 }
245    }
246
247    /// Bytes written so far.
248    pub fn len(&self) -> usize {
249        self.pos
250    }
251
252    /// Appends a `u64`; returns false if it does not fit.
253    pub fn u64(&mut self, v: u64) -> bool {
254        self.raw(&v.to_le_bytes())
255    }
256
257    /// Appends a length-prefixed byte string; returns false if it does not fit.
258    pub fn bytes(&mut self, b: &[u8]) -> bool {
259        self.pos + 4 + b.len() <= self.buf.len() && self.raw(&(b.len() as u32).to_le_bytes()) && self.raw(b)
260    }
261
262    fn raw(&mut self, b: &[u8]) -> bool {
263        if self.pos + b.len() > self.buf.len() {
264            return false;
265        }
266        self.buf[self.pos..self.pos + b.len()].copy_from_slice(b);
267        self.pos += b.len();
268        true
269    }
270
271    /// Appends an attribute record.
272    pub fn attr(&mut self, a: &Attr) -> bool {
273        self.u64(a.ino)
274            && self.u64(a.mode as u64 | (a.nlink as u64) << 32)
275            && self.u64(a.uid as u64 | (a.gid as u64) << 32)
276            && self.u64(a.rdev)
277            && self.u64(a.size)
278            && self.u64(a.blocks)
279            && self.time(a.atime)
280            && self.time(a.mtime)
281            && self.time(a.ctime)
282            && self.u64(a.generation)
283            && self.u64(a.birth_txg)
284    }
285
286    fn time(&mut self, t: (i64, u32)) -> bool {
287        self.u64(t.0 as u64) && self.u64(t.1 as u64)
288    }
289
290    /// Appends an attribute change record (a presence mask, then the values).
291    pub fn setattr(&mut self, s: &SetAttr) -> bool {
292        let mask = s.mode.is_some() as u64
293            | (s.uid.is_some() as u64) << 1
294            | (s.gid.is_some() as u64) << 2
295            | (s.size.is_some() as u64) << 3
296            | (s.atime.is_some() as u64) << 4
297            | (s.mtime.is_some() as u64) << 5;
298        self.u64(mask)
299            && self.u64(s.mode.unwrap_or(0) as u64)
300            && self.u64(s.uid.unwrap_or(0) as u64)
301            && self.u64(s.gid.unwrap_or(0) as u64)
302            && self.u64(s.size.unwrap_or(0))
303            && self.time(s.atime.unwrap_or((0, 0)))
304            && self.time(s.mtime.unwrap_or((0, 0)))
305    }
306
307    /// Appends a directory entry.
308    pub fn entry(&mut self, e: &ReaddirEntry) -> bool {
309        self.u64(e.ino) && self.u64(e.dtype as u64) && self.u64(e.next_cookie) && self.bytes(&e.name)
310    }
311
312    /// Appends an inotify event.
313    pub fn event(&mut self, e: &Event) -> bool {
314        self.u64(e.ino) && self.u64(e.mask as u64 | (e.cookie as u64) << 32) && self.bytes(&e.name)
315    }
316}

318/// Decodes what a [`Writer`] encoded.
319pub struct Reader<'a> {
320    buf: &'a [u8],
321    pos: usize,
322}
323
324impl<'a> Reader<'a> {
325    /// Reads from the start of `buf`.
326    pub fn new(buf: &'a [u8]) -> Self {
327        Self { buf, pos: 0 }
328    }
329
330    /// Next `u64` (0 past the end).
331    pub fn u64(&mut self) -> u64 {
332        let v = self.buf.get(self.pos..self.pos + 8).map_or(0, |b| u64::from_le_bytes(b.try_into().unwrap()));
333        self.pos += 8;
334        v
335    }
336
337    /// Next byte string (empty past the end).
338    pub fn bytes(&mut self) -> &'a [u8] {
339        let len = self.buf.get(self.pos..self.pos + 4).map_or(0, |b| u32::from_le_bytes(b.try_into().unwrap()) as usize);
340        self.pos += 4;
341        let b = self.buf.get(self.pos..self.pos + len).unwrap_or(&[]);
342        self.pos += len;
343        b
344    }
345
346    fn time(&mut self) -> (i64, u32) {
347        (self.u64() as i64, self.u64() as u32)
348    }
349
350    /// Next attribute record.
351    pub fn attr(&mut self) -> Attr {
352        let ino = self.u64();
353        let mode_nlink = self.u64();
354        let ids = self.u64();
355        Attr {
356            ino,
357            mode: mode_nlink as u32,
358            nlink: (mode_nlink >> 32) as u32,
359            uid: ids as u32,
360            gid: (ids >> 32) as u32,
361            rdev: self.u64(),
362            size: self.u64(),
363            blocks: self.u64(),
364            atime: self.time(),
365            mtime: self.time(),
366            ctime: self.time(),
367            generation: self.u64(),
368            birth_txg: self.u64(),
369        }
370    }
371
372    /// Next attribute change record.
373    pub fn setattr(&mut self) -> SetAttr {
374        let mask = self.u64();
375        let (mode, uid, gid, size) = (self.u64() as u32, self.u64() as u32, self.u64() as u32, self.u64());
376        let (atime, mtime) = (self.time(), self.time());
377        SetAttr {
378            mode: (mask & 1 != 0).then_some(mode),
379            uid: (mask & 2 != 0).then_some(uid),
380            gid: (mask & 4 != 0).then_some(gid),
381            size: (mask & 8 != 0).then_some(size),
382            atime: (mask & 16 != 0).then_some(atime),
383            mtime: (mask & 32 != 0).then_some(mtime),
384        }
385    }
386
387    /// Next directory entry.
388    pub fn entry(&mut self) -> ReaddirEntry {
389        let ino = self.u64();
390        let dtype = self.u64() as u8;
391        let next_cookie = self.u64();
392        ReaddirEntry { name: self.bytes().to_vec(), ino, dtype, next_cookie }
393    }
394
395    /// Next inotify event.
396    pub fn event(&mut self) -> Event {
397        let ino = self.u64();
398        let mc = self.u64();
399        Event { ino, mask: mc as u32, cookie: (mc >> 32) as u32, name: self.bytes().to_vec() }
400    }
