1//! # Hydra e1000 network driver
2//!
3//! Drives an Intel 82540EM (e1000) network adapter from Ring 3, in an address space of its
4//! own: its registers are the device memory the supervisor granted (PCI `BAR0`), its
5//! descriptor rings and packet buffers DMA memory it allocates. The kernel enumerates PCI
6//! (enabling bus mastering) and starts it with `bar0=<phys> irq=<line>`.
7//!
8//! The driver only answers requests. When the adapter interrupts, the kernel masks the line
9//! and wakes its network task, which asks for the received frames; the driver acknowledges
10//! the interrupt causes and unmasks the line then.
11//!
12//! Requests on [`ENDPOINT_NET_DEV`] (frames travel through the service window):
13//! - `MSG_NET_TRANSMIT [len]`: send the frame at the start of the window;
14//! - `MSG_NET_RECEIVE`: fill the window with received frames, each a 16-bit little-endian
15//!   length then its bytes; reply `[0, frame count]`;
16//! - `MSG_NET_GET_MAC`: reply `[0, MAC bytes 0..4, MAC bytes 4..6]`;
17//! - `MSG_NET_GET_STATUS`: reply `[0, link up]`.
18
19#![no_std]
20#![no_main]

31const REG_CTRL: usize = 0x0000;
32const REG_STATUS: usize = 0x0008;
33const REG_ICR: usize = 0x00C0;
34const REG_IMS: usize = 0x00D0;
35const REG_IMC: usize = 0x00D8;
36const REG_RDTR: usize = 0x2820;
37const REG_RCTL: usize = 0x0100;
38const REG_TCTL: usize = 0x0400;
39const REG_TIPG: usize = 0x0410;
40const REG_RDBAL: usize = 0x2800;
41const REG_RDBAH: usize = 0x2804;
42const REG_RDLEN: usize = 0x2808;
43const REG_RDH: usize = 0x2810;
44const REG_RDT: usize = 0x2818;
45const REG_TDBAL: usize = 0x3800;
46const REG_TDBAH: usize = 0x3804;
47const REG_TDLEN: usize = 0x3808;
48const REG_TDH: usize = 0x3810;
49const REG_TDT: usize = 0x3818;
50const REG_RAL0: usize = 0x5400;
51const REG_RAH0: usize = 0x5404;
52
53const CTRL_SLU: u32 = 1 << 6;
54const CTRL_RST: u32 = 1 << 26;
55const STATUS_LU: u32 = 1 << 1;
56const RCTL_EN: u32 = 1 << 1;
57const RCTL_SBP: u32 = 1 << 2;
58const RCTL_UPE: u32 = 1 << 3;
59const RCTL_MPE: u32 = 1 << 4;
60const RCTL_BAM: u32 = 1 << 15;
61const RCTL_SECRC: u32 = 1 << 26;
62const TCTL_EN: u32 = 1 << 1;
63const TCTL_PSP: u32 = 1 << 3;
64
65const INT_LSC: u32 = 1 << 2;
66const INT_RXDMT0: u32 = 1 << 4;
67const INT_RXO: u32 = 1 << 6;
68const INT_RXT0: u32 = 1 << 7;

70const NUM_TX: usize = 32;
71const NUM_RX: usize = 256;
72const BUF: usize = 2048;
73/// Size of the register window mapped.
74const MMIO_LEN: usize = 128 * 1024;
75
76const EIO: i64 = -5;
77const EINVAL: i64 = -22;

79/// A legacy transmit or receive descriptor (16 bytes; the fields differ by direction).
80#[repr(C)]
81#[derive(Clone, Copy)]
82struct Desc {
83    addr: u64,
84    length: u16,
85    /// TX: checksum offset; RX: checksum.
86    cso_or_csum_lo: u8,
87    /// TX: command; RX: checksum high byte.
88    cmd_or_csum_hi: u8,
89    status: u8,
90    /// TX: checksum start; RX: errors.
91    css_or_errors: u8,
92    special: u16,
93}
94
95struct Nic {
96    regs: *mut u8,
97    /// DMA memory: RX descriptors, TX descriptors, RX buffers, TX buffers.
98    mem: *mut u8,
99    mem_phys: u64,
100    mac: [u8; 6],
101    tx_cur: usize,
102    rx_cur: usize,
103    irq: Option<u8>,
104}
105
106const RX_DESC_OFF: usize = 0;
107const TX_DESC_OFF: usize = NUM_RX * 16;
108const RX_BUF_OFF: usize = 8192;
109const TX_BUF_OFF: usize = RX_BUF_OFF + NUM_RX * BUF;
110const DMA_LEN: usize = TX_BUF_OFF + NUM_TX * BUF;
111
112impl Nic {
113    fn read(&self, reg: usize) -> u32 {
114        unsafe { core::ptr::read_volatile(self.regs.add(reg) as *const u32) }
115    }
116
117    fn write(&self, reg: usize, v: u32) {
118        unsafe { core::ptr::write_volatile(self.regs.add(reg) as *mut u32, v) }
119    }
120
121    fn rx_desc(&self, i: usize) -> *mut Desc {
122        unsafe { self.mem.add(RX_DESC_OFF + i * 16) as *mut Desc }
123    }
124
125    fn tx_desc(&self, i: usize) -> *mut Desc {
126        unsafe { self.mem.add(TX_DESC_OFF + i * 16) as *mut Desc }
127    }
128
129    /// Resets the adapter and sets up both rings.
130    fn init(&mut self) {
131        self.write(REG_IMC, 0xFFFF_FFFF);
132        self.write(REG_CTRL, self.read(REG_CTRL) | CTRL_RST);
133        for _ in 0..100_000 {
134            core::hint::spin_loop();
135        }
136        self.write(REG_IMC, 0xFFFF_FFFF);
137        let (ral, rah) = (self.read(REG_RAL0), self.read(REG_RAH0));
138        self.mac = [ral as u8, (ral >> 8) as u8, (ral >> 16) as u8, (ral >> 24) as u8, rah as u8, (rah >> 8) as u8];
139
140        for i in 0..NUM_TX {
141            unsafe {
142                self.tx_desc(i).write_volatile(Desc {
143                    addr: self.mem_phys + (TX_BUF_OFF + i * BUF) as u64,
144                    length: 0,
145                    cso_or_csum_lo: 0,
146                    cmd_or_csum_hi: 0,
147                    status: 1, // free
148                    css_or_errors: 0,
149                    special: 0,
150                })
151            };
152        }
153        for i in 0..NUM_RX {
154            unsafe {
155                self.rx_desc(i).write_volatile(Desc {
156                    addr: self.mem_phys + (RX_BUF_OFF + i * BUF) as u64,
157                    length: 0,
158                    cso_or_csum_lo: 0,
159                    cmd_or_csum_hi: 0,
160                    status: 0,
161                    css_or_errors: 0,
162                    special: 0,
163                })
164            };
165        }
166        let tx = self.mem_phys + TX_DESC_OFF as u64;
167        self.write(REG_TDBAL, tx as u32);
168        self.write(REG_TDBAH, (tx >> 32) as u32);
169        self.write(REG_TDLEN, (NUM_TX * 16) as u32);
170        self.write(REG_TDH, 0);
171        self.write(REG_TDT, 0);
172        self.write(REG_TCTL, TCTL_EN | TCTL_PSP | (15 << 4) | (0x40 << 12));
173        self.write(REG_TIPG, 0x0060_200A);
174
175        let rx = self.mem_phys + RX_DESC_OFF as u64;
176        self.write(REG_RDBAL, rx as u32);
177        self.write(REG_RDBAH, (rx >> 32) as u32);
178        self.write(REG_RDLEN, (NUM_RX * 16) as u32);
179        self.write(REG_RDH, 0);
180        self.write(REG_RDT, (NUM_RX - 1) as u32);
181        self.write(REG_RCTL, RCTL_EN | RCTL_SBP | RCTL_UPE | RCTL_MPE | RCTL_BAM | RCTL_SECRC);
182
183        self.write(REG_CTRL, self.read(REG_CTRL) | CTRL_SLU);
184        for _ in 0..50 {
185            if self.link_up() {
186                break;
187            }
188            sys::sleep_ms(10);
189        }
190    }
191
192    fn link_up(&self) -> bool {
193        self.read(REG_STATUS) & STATUS_LU != 0
194    }
195
196    /// Routes the adapter's interrupt to this driver and enables the causes the kernel's
197    /// network task waits for.
198    fn enable_interrupts(&mut self, irq: u8) {
199        if sys::irq_register(irq).is_err() {
200            println!("[e1000-driver] interrupt line {} unavailable; the network is polled", irq);
201            return;
202        }
203        self.irq = Some(irq);
204        self.write(REG_RDTR, 0); // an interrupt per frame
205        self.read(REG_ICR);
206        self.write(REG_IMS, INT_RXT0 | INT_RXO | INT_RXDMT0 | INT_LSC);
207    }
208
209    fn transmit(&mut self, frame: &[u8]) -> Result<(), &'static str> {
210        if frame.is_empty() || frame.len() > BUF {
211            return Err("bad frame length");
212        }
213        let cur = self.tx_cur;
214        unsafe {
215            core::ptr::copy_nonoverlapping(frame.as_ptr(), self.mem.add(TX_BUF_OFF + cur * BUF), frame.len());
216            let d = self.tx_desc(cur);
217            let mut desc = d.read_volatile();
218            desc.length = frame.len() as u16;
219            desc.cmd_or_csum_hi = (1 << 0) | (1 << 1) | (1 << 3); // EOP | IFCS | RS
220            desc.status = 0;
221            d.write_volatile(desc);
222        }
223        self.tx_cur = (cur + 1) % NUM_TX;
224        self.write(REG_TDT, self.tx_cur as u32);
225        for _ in 0..1_000_000 {
226            if unsafe { self.tx_desc(cur).read_volatile().status } & 1 != 0 {
227                return Ok(());
228            }
229            core::hint::spin_loop();
230        }
231        Err("transmit timed out")
232    }
233
234    /// Moves received frames into `out` (length-prefixed); returns how many. Acknowledges
235    /// the interrupt and unmasks the line.
236    fn receive(&mut self, out: &mut [u8]) -> usize {
237        self.read(REG_ICR); // acknowledge: the adapter stops asserting the line
238        let (mut count, mut at) = (0, 0);
239        loop {
240            let d = self.rx_desc(self.rx_cur);
241            let mut desc = unsafe { d.read_volatile() };
242            if desc.status & 1 == 0 {
243                break;
244            }
245            let len = desc.length as usize;
246            if at + 2 + len > out.len() {
247                break; // the rest waits for the next request
248            }
249            out[at..at + 2].copy_from_slice(&(len as u16).to_le_bytes());
250            unsafe { core::ptr::copy_nonoverlapping(self.mem.add(RX_BUF_OFF + self.rx_cur * BUF), out[at + 2..].as_mut_ptr(), len) };
251            at += 2 + len;
252            count += 1;
253            desc.status = 0;
254            unsafe { d.write_volatile(desc) };
255            self.write(REG_RDT, self.rx_cur as u32);
256            self.rx_cur = (self.rx_cur + 1) % NUM_RX;
257        }
258        if let Some(irq) = self.irq {
259            let _ = sys::irq_ack(irq);
260        }
261        count
262    }
263}
264
265fn reply(args: [u64; 4]) -> Message {
266    Message::new(MSG_NET_REPLY as u64, args)
267}

269/// Value of a `key=<number>` argument (decimal or `0x` hex).
270fn arg(args: &[&str], key: &str) -> Option<u64> {
271    let v = args.iter().find_map(|a| a.strip_prefix(key)?.strip_prefix('='))?;
272    match v.strip_prefix("0x") {
273        Some(h) => u64::from_str_radix(h, 16).ok(),
274        None => v.parse().ok(),
275    }
276}

278fn main(args: &[&str]) {
279    let Some(bar0) = arg(args, "bar0") else {
280        println!("[e1000-driver] no adapter given (bar0=...)");
281        return;
282    };
283    let regs = match sys::physmap(bar0, MMIO_LEN) {
284        Ok(r) => r,
285        Err(e) => {
286            println!("[e1000-driver] cannot map the registers at {:#x}: error {}", bar0, e);
287            return;
288        }
289    };
290    let (mem, mem_phys) = match sys::dma_alloc(DMA_LEN) {
291        Ok(m) => m,
292        Err(e) => {
293            println!("[e1000-driver] no DMA memory: error {}", e);
294            return;
295        }
296    };
297    let mut nic = Nic { regs, mem, mem_phys, mac: [0; 6], tx_cur: 0, rx_cur: 0, irq: None };
298    nic.init();
299    if let Some(irq) = arg(args, "irq") {
300        nic.enable_interrupts(irq as u8);
301    }
302    let mut server = match Server::register(ENDPOINT_NET_DEV) {
303        Ok(s) => s,
304        Err(e) => {
305            println!("[e1000-driver] cannot register on endpoint {:#x}: error {}", ENDPOINT_NET_DEV, e);
306            return;
307        }
308    };
309    let m = nic.mac;
310    println!(
311        "[e1000-driver] MAC {:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}, link {}",
312        m[0], m[1], m[2], m[3], m[4], m[5],
313        if nic.link_up() { "up" } else { "down" }
314    );
315    server.serve(|srv, msg| match msg.label as u32 {
316        MSG_NET_TRANSMIT => {
317            let len = msg.args[0] as usize;
318            if len == 0 || len > BUF.min(WINDOW_SIZE) {
319                return reply([EINVAL as u64, 0, 0, 0]);
320            }
321            match nic.transmit(&srv.window()[..len]) {
322                Ok(()) => reply([0, len as u64, 0, 0]),
323                Err(_) => reply([EIO as u64, 0, 0, 0]),
324            }
325        }
326        MSG_NET_RECEIVE => {
327            let n = nic.receive(srv.window());
328            reply([0, n as u64, 0, 0])
329        }
330        MSG_NET_GET_MAC => {
331            let m = nic.mac;
332            reply([0, u32::from_le_bytes([m[0], m[1], m[2], m[3]]) as u64, u16::from_le_bytes([m[4], m[5]]) as u64, 0])
333        }
334        MSG_NET_GET_STATUS => {
335            let d = unsafe { nic.rx_desc(nic.rx_cur).read_volatile() };
336            reply([0, nic.link_up() as u64, nic.read(REG_RDH) as u64 | (nic.read(REG_RDT) as u64) << 16, nic.rx_cur as u64 | (d.status as u64) << 16 | (d.length as u64) << 32 | (d.css_or_errors as u64) << 48])
337        }
338        _ => reply([(-38i64) as u64, 0, 0, 0]),
339    })
